\sechead{1}{Elements integral over a ring}{ELEMENTS INTEGRAL OVER A RING}

\begin{theorem}
\label{thm:II.1.1}
Let $R$ be a ring, $A$ a subring of $R$, and $x$ an element of $R$. The
following statements are equivalent:
\begin{alist}
\item There exist $a_{0},\ldots,a_{n-1}\in A$ such that
\begin{equation}
x^{n}+a_{n-1}x^{n-1}+\cdots+a_{1}x+a_{0}=0
\label{eq:II.1.1}
\end{equation}
(i.e.\ $x$ is a root of a monic polynomial with coefficients in $A$).
\item The ring $A[x]$ is an $A$-module of finite type.
\item There exists a subring $B$ of $R$ which contains $A$ and $x$ and which
is an $A$-module of finite type.
\end{alist}
\end{theorem}

\begin{proof}
Call $M$ the $A$-submodule of $R$ generated by $1,x,\ldots,x^{n-1}$. By (a)
$x^{n}\in M$. Multiplying \eqref{eq:II.1.1} by $x^{j}$, we obtain
$x^{n+j}=-a_{n-1}x^{n+j-1}-\cdots-a_{0}x^{j}$. In fact, induction on $j$
implies that $x^{n+j}\in M$, for all $j\geq0$. As $A[x]$ is the $A$-module
generated by the $x^{k}$ ($k\geq0$), we see that $A[x]=M$. Thus (a) implies
(b). That (b) implies (c) is clear. Let us show that (c) implies (a). Let
$(y_{1},\ldots,y_{n})$ be a finite set of generators for $B$ as a module
over $A$, i.e.\ $B=Ay_{1}+\cdots+Ay_{n}$. Since $x\in B$ and since $B$ is a
subring of $R$, it follows that $xy_{i}\in B$ for all $i=1,\ldots,n$.
Therefore,
\[
xy_{i}=\sum_{j=1}^{n}a_{ij}y_{j},\quad\text{for any }i=1,\ldots,n,\quad
a_{ij}\in A,\quad1\leq i,\ j\leq n.
\]
This means that
\[
\sum_{j=1}^{n}\left(\delta_{ij}x-a_{ij}\right)y_{j}=0,
\quad i=1,\ldots,n.
\]
Consider this system of $n$ homogeneous linear equations in
$(y_{1},\ldots,y_{n})$. Write $d$ for the determinant
$\det(\delta_{ij}x-a_{ij})$. The calculation leading to Cramer's rule shows
that $dy_{i}=0$ for all $i$. This means that $db=0$ for all $b\in B$; in
particular, $d\cdot1=0$, so $d=0$. But $d$ is clearly a monic polynomial in
$x$, since the highest order term appears in the expansion of the product
$\prod_{i=1}^{n}(x-a_{ii})$ of the entries on the principal diagonal. Thus
(c) implies (a).
\end{proof}

\begin{definition}\label{def:II.1.1}
Let $R$ be a ring and let $A$ be a subring of $R$. An element $x$ of $R$ is
called \emph{integral over} $A$ if it satisfies the equivalent conditions
(a), (b), and (c) of Theorem \ref{thm:II.1.1}. Let $P\in A[X]$ be a monic
polynomial such that $P(x)=0$ ((a) implies that such a polynomial exists).
The relation $P(x)=0$ is called an \emph{equation of integral dependence} of
$x$ over $A$.
\end{definition}

\begin{statement}{Example}
The element $x=\sqrt{2}$ of $\mathbf{R}$ is integral over $\mathbf{Z}$. The
relation $x^{2}-2=0$ is an equation of integral dependence.
\end{statement}

\begin{proposition}
\label{prop:II.1.1}
Let $R$ be a ring, $A$ a subring of $R$, and let $(x_{i})_{1\leq i\leq n}$ be
a finite set of elements of $R$. If, for all $i$, $x_{i}$ is integral over
$A[x_{1},\ldots,x_{i-1}]$ (in particular if all the $x_{i}$'s are integral
over $A$), then $A[x_{1},\ldots,x_{n}]$ is an $A$-module of finite type.
\end{proposition}

\begin{proof}
We argue by induction on $n$. For $n=1$, we have a repeat of assertion (b)
of Theorem \ref{thm:II.1.1}. Assume that $B=A[x_{1},\ldots,x_{n-1}]$ is an
$A$-module of finite type. Then $B=\sum_{j=1}^{p}Ab_{j}$. The case $n=1$
implies that $A[x_{1},\ldots,x_{n}]=B[x_{n}]$ is a $B$-module of finite
type. Write $B[x_{n}]=\sum_{k=1}^{q}Bc_{k}$. Then
\[
A[x_{1},\ldots,x_{n}]=\sum_{k=1}^{q}Bc_{k}
=\sum_{k=1}^{q}\Bigl(\sum_{j=1}^{p}Ab_{j}\Bigr)c_{k}
=\sum_{j,k}Ab_{j}c_{k}.
\]
Thus $(b_{j}c_{k})_{\substack{1\leq j\leq p\\1\leq k\leq q}}$ is a finite
set of generators for $A[x_{1},\ldots,x_{n}]$ as a module over $A$.
\end{proof}

\begin{corollary}
\label{cor:II.1.1}
Let $R$ be a ring, $A$ a subring of $R$, $x$ and $y$ elements of $R$ which
are integral over $A$. Then $x+y$, $x-y$, and $xy$ are integral over $A$.
\end{corollary}

\begin{proof}
Clearly $x+y$, $x-y$, and $xy\in A[x,y]$. According to Proposition
\ref{prop:II.1.1}, $A[x,y]$ is an $A$-module of finite type. According to
part (c) of Theorem \ref{thm:II.1.1}, $x+y$, $x-y$, and $xy$ are integral
over $A$.
\end{proof}

\begin{corollary}
\label{cor:II.1.2}
Let $R$ be a ring and let $A$ be a subring of $R$. The set $A'$ of elements
of $R$ which are integral over $A$ is a subring of $R$ which contains $A$.
\end{corollary}

\begin{proof}
Corollary \ref{cor:II.1.1} implies that $A'$ is a subring of $R$. We have
$A\subset A'$, since, if $a\in A$, $a$ is a root of the monic polynomial
$P(X)=X-a$, which has coefficients in $A$.
\end{proof}

\begin{definition}\label{def:II.1.2}
Let $R$ be a ring, $A$ a subring of $R$. The ring $A'$ of elements of $R$
which are integral over $A$ is called the \emph{integral closure} of $A$ in
$R$. Let $A$ be an integral domain and let $K$ be its field of fractions. The
integral closure of $A$ in $K$ is called the \emph{integral closure of} $A$.
Let $B$ be a ring and $A$ a subring of $B$. We say that $B$ is
\emph{integral over} $A$ if every element of $B$ is integral over $A$ (i.e.\
if the integral closure of $A$ in $B$ is $B$ itself).
\end{definition}

\begin{proposition}[Transitivity]
\label{prop:II.1.2}
Let $C$ be a ring, $B$ a subring of $C$, and $A$ a subring of $B$. If $B$ is
integral over $A$ and if $C$ is integral over $B$, then $C$ is integral over
$A$.
\end{proposition}

\begin{proof}
Let $x\in C$. Then $x$ is integral over $B$, so there is an equation of
integral dependence $x^{n}+b_{n-1}x^{n-1}+\cdots+b_{0}=0$ with
$b_{i}\in B$, $i=0,\ldots,n-1$. Put $B'=A[b_{0},\ldots,b_{n-1}]$. Then $x$
is integral over $B'$. As $B$ is integral over $A$, the $b_{i}$ are integral
over $A$. Therefore, Proposition \ref{prop:II.1.1} implies that
$B'[x]=A[b_{0},\ldots,b_{n-1},x]$ is an $A$-module of finite type. By part
(c) of Theorem \ref{thm:II.1.1}, $x$ is integral over $A$. Thus $C$ is
integral over $A$.
\end{proof}

\begin{proposition}
\label{prop:II.1.3}
Let $B$ be an integral domain and $A$ a subring of $B$ such that $B$ is
integral over $A$. In order that $B$ be a field it is necessary and
sufficient that $A$ be a field.
\end{proposition}

\begin{proof}
Suppose that $A$ is a field and let $b\in B$, $b\neq0$. Then $A[b]$ is a
finite dimensional vector space over $A$ (part (b) of Theorem
\ref{thm:II.1.1}). On the other hand $y\mapsto by$ is an $A$-linear
transformation of $A[b]$. It is injective since $A[b]$ is an integral domain
and since $b\neq0$. Therefore, it is surjective. There exists $b'\in A[b]$
such that $bb'=1$. This means that, for any $b\in B-(0)$, $b$ is invertible
in $B$, so $B$ is a field.\footnote{The same reasoning, involving the
mapping $y\mapsto by$, allows one to conclude that any \emph{finite}
integral domain is a field.}

Conversely, suppose that $B$ is a field. Let $a\in A-(0)$. Then $a$ has an
inverse $a^{-1}\in B$ which satisfies an equation of integral dependence
\[
a^{-n}+a_{n-1}a^{-n+1}+\cdots+a_{1}a^{-1}+a_{0}=0,\quad a_{i}\in A.
\]
Multiplying by $a^{n-1}$, we obtain
\[
a^{-1}=-\left(a_{n-1}+\cdots+a_{1}a^{n-2}+a_{0}a^{n-1}\right),
\]
which shows that $a^{-1}\in A$. Thus $A$ is a field.
\end{proof}

\sechead{2}{Integrally closed rings}{INTEGRALLY CLOSED RINGS}

\begin{statement}{Definition}
A ring $A$ is called \emph{integrally closed} if it is an integral domain and
if it is its own integral closure.
\end{statement}

In other words, every element $x$ of the field of fractions $K$ of $A$ which
is integral over $A$ belongs to $A$.

\begin{example}\label{exa:II.2.1}
Let $A$ be an integral domain and let $K$ be its field of fractions. Then the
integral closure $A'$ of $A$ (i.e.\ the integral closure of $A$ in $K$) is
integrally closed. This follows from the fact that the integral closure of
$A'$ is integral over $A'$, therefore over $A$ (\hyperref[sec:2.1]{§1}, Proposition
\ref{prop:II.1.2}). It therefore equals $A'$.
\end{example}

\begin{example}\label{exa:II.2.2}
Every principal ideal ring is integrally closed.
\end{example}

\begin{proof}
By definition a principal ideal ring is an integral domain. Let $x$ be an
element of the field of fractions of $A$ which is integral over $A$. Let
\begin{equation}
x^{n}+a_{n-1}x^{n-1}+\cdots+a_{1}x+a_{0}=0\quad(a_{i}\in A)
\label{eq:II.2.1}
\end{equation}
be an integral dependence equation for $x$ over $A$. Write $x=a/b$ with $a$
and $b$ relatively prime elements of $A$. Substitute in \eqref{eq:II.2.1} and
multiply through by $b^{n}$ to obtain
\[
a^{n}+b\left(a_{n-1}a^{n-1}+\cdots+a_{1}ab^{n-2}+a_{0}b^{n-1}\right)=0.
\]
Thus $b$ divides $a^{n}$. As $b$ is relatively prime to $a$, repeated
application of Euclid's lemma shows that $b$ divides $a$. Therefore, $b$ is a
unit in $A$. Thus, $x=a/b\in A$ and $A$ is integrally closed.
\end{proof}

One may observe that only the multiplicative properties of principal ideal
rings have been used (relative primeness, Euclid's lemma). The same argument
thus shows that every factorial ring is integrally closed.

\sechead{3}{Elements algebraic over a field. Algebraic extensions}%
        {ELEMENTS ALGEBRAIC OVER A FIELD. ALGEBRAIC EXTENSIONS}

\begin{statement}{Definition}\phantomsection\label{def:II.3.1}
Let $R$ be a ring and $K$ a subfield of $R$. An element $x\in R$ is called
\emph{algebraic over} $K$ if there exist elements $a_{0},\ldots,a_{n}\in K$,
not all of which are zero, such that
$a_{n}x^{n}+\cdots+a_{1}x+a_{0}=0$.
\end{statement}

Equivalently, the monomials $(x^{j})_{j\in\mathbf{N}}$ are linearly
dependent over $K$. An element of $R$ which is not algebraic over $K$ is
called \emph{transcendental over} $K$; i.e.\ $x$ is transcendental over $K$
if and only if the monomials $(x^{j})_{j\in\mathbf{N}}$ are linearly
independent over $K$.

In the relation of \hyperref[def:II.3.1]{Definition 1}, we may assume that $a_{n}\neq0$. In this
case $a_{n}^{-1}\in K$; multiplying through by $a_{n}^{-1}$ we obtain an
equation of integral dependence. Therefore:
\begin{equation}
\text{Over a field, algebraic}=\text{integral.}
\label{eq:II.3.1}
\end{equation}

We may thus apply the theory of integral elements. For example, for
$K\subset\mathbf{R}$ and $x\in\mathbf{R}$, Theorem \ref{thm:II.1.1}, (b) of
\hyperref[sec:2.1]{§1} asserts:
\begin{equation}
x\text{ algebraic over }K\Leftrightarrow[K[x]:K]\text{ finite.}
\label{eq:II.3.2}
\end{equation}

We say that a ring $R$ containing a field $K$ is \emph{algebraic over} $K$ if
every element of $R$ is algebraic over $K$. If $R$ is a field, then $R$ is
called an \emph{algebraic extension} of $K$.

Given a field $L$ and a subfield $K$ of $L$, we call the dimension $[L:K]$
the \emph{degree} of $L$ over $K$. In this context Theorem \ref{thm:II.1.1},
(c) of \hyperref[sec:2.1]{§1} has the following interpretation:
\begin{equation}
\text{If the degree of }L\text{ over }K\text{ is finite, }L\text{ is an
algebraic extension of }K.
\label{eq:II.3.3}
\end{equation}

Any extension field of finite degree over $\mathbf{Q}$ is called an
\emph{algebraic number field} (or, simply, a \emph{number field}).

\begin{proposition}
\label{prop:II.3.1}
Let $K$ be a field, $L$ an algebraic extension of $K$, and $M$ an algebraic
extension of $L$. Then $M$ is an algebraic extension of $L$. Furthermore,
$[M:K]=[M:L][L:K]$ (``multiplicativity of degrees'').
\end{proposition}

\begin{proof}
The first assertion is a special case of Proposition \ref{prop:II.1.2}.
Moreover, if $(x_{i})_{i\in I}$ is a base of $L$ over $K$ and
$(y_{j})_{j\in J}$ is a base for $M$ over $L$, then
$(x_{i}y_{j})_{(i,j)\in I\times J}$ is a base for $M$ over $K$. As in
Proposition \ref{prop:II.1.1} we see that
$(x_{i}y_{j})_{(i,j)\in I\times J}$ generates $M$ over $K$. A relation
$\sum a_{ij}x_{i}y_{j}=0$ with $a_{ij}\in K$ entails
$\sum(\sum a_{ij}x_{i})y_{j}=0$, whence $\sum a_{ij}x_{i}=0$ for all $j$
(since $\sum a_{ij}x_{i}\in L$), and consequently $a_{ij}=0$ for all
$(i,j)\in I\times J$. This proves that $[M:K]=[M:L][L:K]$.
\end{proof}

\begin{proposition}
\label{prop:II.3.2}
Let $R$ be a ring and $K$ a subfield of $R$. Then:
\begin{alist}
\item The set $K'$ of elements of $R$ algebraic over $K$ is a subring of $R$
containing $K$.
\item If $R$ is an integral domain, $K'$ is a subfield of $R$.
\end{alist}
\end{proposition}

\begin{proof}
(a) is a special case of Corollary \ref{cor:II.1.2} of Proposition
\ref{prop:II.1.1}, \hyperref[sec:2.1]{§1}, and (b) follows from Proposition \ref{prop:II.1.3} of
\hyperref[sec:2.1]{§1}.
\end{proof}

Now we study the elements algebraic over a field in greater detail. Let $R$
be a ring, $K$ a subfield of $R$, and let $x$ be an element of $R$. Write
$K[X]$ for the ring of polynomials in one variable over $K$. There exists a
unique homomorphism $\varphi:K[X]\to R$ such that $\varphi(X)=x$ and such
that $\varphi(a)=a$ for all $a\in K$. The image of $\varphi$ is $K[x]$. The
definition of algebraic element may be reformulated as follows:
\begin{equation}
\text{An element }x\text{ is algebraic over }K\Leftrightarrow
\operatorname{Ker}(\varphi)\neq(0).
\label{eq:II.3.4}
\end{equation}

\begin{proof}
If $x$ is transcendental over $K$, then obviously
$\operatorname{Ker}(\varphi)=(0)$. In any case the ideal
$\operatorname{Ker}(\varphi)$ is a principal ideal $(F(X))$ (since $K[X]$ is
a principal ideal ring). In the case that $x$ is algebraic over $K$, it is
generated by a non-zero polynomial $F(X)$.
\end{proof}

\begin{statement}{Remark}
We may assume that $F(X)$ is monic, since $K$ is a field. $F(X)$ is then
uniquely determined by $K$ and $x$; we call it the \emph{minimal polynomial}
of $x$ over $K$. Its properties are as follows:
\begin{equation}
\begin{gathered}
\text{Let }F(X)\text{ be the minimal polynomial of }x\text{ over }K.
\text{ Let }G(X)\in K[X].\ G(x)=0\text{ if and only if }\\
\F(X)\text{ divides }G(X)\text{ in }K[X].
\end{gathered}
\label{eq:II.3.5}
\end{equation}
\end{statement}

Passing to the quotient ring, we obtain a canonical isomorphism:
\begin{equation}
K[X]/(F(X))\simeq K[x].
\label{eq:II.3.6}
\end{equation}

With the same notations, suppose now that $x$ is algebraic over $K$ and let
$F(X)$ be its minimal polynomial. Applying \eqref{eq:II.3.5} and Proposition
\ref{prop:II.1.3} of \hyperref[sec:2.1]{§1}, we obtain the equivalence of the following
statements:
\begin{equation}
\begin{gathered}
K[x]\text{ is a field }\Leftrightarrow\ K[x]\text{ is an integral domain}\\
\Leftrightarrow\ F(X)\text{ is irreducible.}
\end{gathered}
\label{eq:II.3.7}
\end{equation}

On the other hand, if $K$ is a field and $F(X)\in K[X]$ is irreducible, then
$K[X]/(F(X))$ is a field containing $K$ and, writing $x$ for the projection
of $X\in K[X]$ into this field, we have $F(x)=0$. Thus $X-x$ divides $F(X)$
in the field $K[x]$. More generally:

\begin{proposition}
\label{prop:II.3.3}
Let $K$ be a field and let $P(X)\in K[X]$ be a nonconstant polynomial. There
exists an algebraic extension of finite degree $K'$ of $K$ such that $P(X)$
factors in $K'[X]$ into a product of polynomials of degree one (linear
polynomials).
\end{proposition}

\begin{proof}
We argue by induction on the degree $d^{\circ}$ of $P(X)$. There is nothing
to prove in the case $d^{\circ}=1$. Let $F(X)$ be an irreducible factor of
$P(X)$. We have just seen that there exists an extension $K''$ of finite
degree over $K$ (e.g.\ $K[X]/(F(X))$) containing an element $x$ such that
$X-x$ divides $F(X)$ in $K''[X]$. Thus $P(X)=(X-x)P_{1}(X)$ with
$P_{1}(X)\in K''[X]$. According to the induction hypothesis $P_{1}(X)$
factors into a product of linear polynomials in an extension $K'$ of finite
degree over $K''$. By Proposition \ref{prop:II.3.1}, $K'$ is of finite
degree over $K$, and $P(X)$ is a product of linear polynomials in $K'[X]$.
\end{proof}

\begin{statement}{Remark. Algebraically closed fields}
A field $K$ is called \emph{algebraically closed} if every non-constant
polynomial $P(X)\in K[X]$ may be expressed as a product of linear factors,
all lying in $K[X]$. For this, as follows by induction on the degree of the
polynomial, it suffices that every non-constant polynomial have a root in
$K$. Extending Proposition \ref{prop:II.3.3} by means of Zorn's lemma (cf.\
\hyperref[bib:1]{[1]}, Chapter V, and \hyperref[bib:9]{[9]}, Chapter II), one may show that any field is
imbeddable in an algebraically closed field.
\end{statement}

One may prove, by means of the techniques of mathematical analysis and in
many different ways,\footnote{For a proof which depends upon the properties of
continuous functions defined on compact topological spaces, see \hyperref[bib:4]{[4]}. For a
proof which makes use of the properties of holomorphic functions of a
complex variable, see \hyperref[bib:3]{[3]}. We will give in an appendix to this chapter a
more algebraic proof, which will involve no analysis beyond the most
elementary properties of the real numbers.} that the field $\mathbf{C}$ of
complex numbers is algebraically closed. We shall not need more than this.

\sechead{4}{Conjugate elements, conjugate fields}%
        {CONJUGATE ELEMENTS, CONJUGATE FIELDS}

Given two fields $L$ and $L'$ both containing a field $K$ we call any
isomorphism $\varphi:L\to L'$ such that $\varphi(a)=a$ for all $a\in K$ a
\emph{$K$-isomorphism of $L$ on $L'$}. In this case we say that $L$ and $L'$
are $K$-isomorphic, or (if they are algebraic over $K$) we say that they are
\emph{conjugate over} $K$. Given two extensions $L$ and $L'$ of $K$, we say
that two elements $x\in L$ and $x'\in L'$ are \emph{conjugate over} $K$ if
there exists a $K$-isomorphism $\varphi:K(x)\to K(x')$ such that
$\varphi(x)=x'$. Such a $\varphi$ is, of course, unique. The existence of a
$\varphi$ means that either $x$ and $x'$ are both transcendental over $K$ or
both are algebraic over $K$ with the same minimal polynomial (cf.\
\eqref{eq:II.3.5}, \hyperref[sec:2.3]{§3}).

\begin{statement}{Example}
Let $F(X)$ be an irreducible polynomial of degree $n$ over $K$ and let
$x_{1},\ldots,x_{n}$ be its roots in an extension $K'$ of $K$ (\hyperref[sec:2.3]{§3},
Proposition \ref{prop:II.3.3}). Then the $x_{i}$'s are pairwise conjugate
over $K$ (\hyperref[sec:2.3]{§3}, \eqref{eq:II.3.6}), and the fields $K[x_{i}]$ are also
pairwise conjugate.
\end{statement}

\begin{statement}{Lemma}
Let $K$ be a field of characteristic zero or a finite field, let
$F(X)\in K[X]$ be a monic irreducible polynomial, and let
$F(X)=\prod_{i=1}^{n}(X-x_{i})$ be its decomposition into a product of
linear factors in an extension $K'$ of $K$ (\hyperref[sec:2.3]{§3}, Proposition
\ref{prop:II.3.3}). Then the $n$ roots $x_{1},\ldots,x_{n}$ of $F(X)$ are
distinct.
\end{statement}

\begin{proof}
If not, $F(X)$ has a root in common with its derivative $F'(X)$. Therefore,
$F(X)$ divides $F'(X)$ (\eqref{eq:II.3.5}, \hyperref[sec:2.3]{§3}). Since
$d^{\circ}F'<d^{\circ}F$, this means that $F'(X)$ is the polynomial zero.
However, $F(X)=X^{n}+a_{n-1}X^{n-1}+\cdots+a_{0}$ ($a_{i}\in K$) and
\[
F'(X)=nX^{n-1}+(n-1)a_{n-1}X^{n-2}+\cdots+a_{1}.
\]
Thus $n\cdot1=0$, $j\cdot a_{j}=0$, $j=1,\ldots,n-1$, which is impossible in
characteristic zero. In characteristic $p\neq0$, these relations imply that
$p$ divides $n$ and that $a_{j}=0$ for all $j$ not divisible by $p$ (recall
that $p$ is a prime number). Thus $F(X)$ is of the form
\[
F(X)=X^{qp}+b_{q-1}X^{(q-1)p}+\cdots+b_{1}X^{p}+b_{0}\quad(b_{i}\in K).
\]
If each of the $b_{i}$'s is a $p$th power, i.e.\ $b_{i}=c_{i}^{p}$ with
$c_{i}\in K$, then
\[
F(X)=\left(X^{q}+c_{q-1}X^{q-1}+\cdots+c_{0}\right)^{p}
\quad(\text{\hyperref[sec:1.7]{Chapter I, §7}, Proposition \ref{prop:I.7.1}}),
\]
and $F(X)$ is not irreducible. But, if $K$ is a finite field with $p\neq0$
its characteristic, the mapping $x\mapsto x^{p}$ of $K$ into $K$ is injective
(since $x^{p}=y^{p}$ implies $x^{p}-y^{p}=0$, so $(x-y)^{p}=0$ which implies
$x-y=0$); it is thus surjective, since $K$ is finite. Therefore, $F(X)$ is
not irreducible and we have a contradiction.
\end{proof}

The fields $K$ of characteristic $p\neq0$ for which $x\mapsto x^{p}$ is
surjective (i.e.\ for which every element of $K$ is a $p$th power) are called
\emph{perfect fields}. We have just shown that finite fields are perfect. By
convention fields of characteristic zero are also considered perfect. The
preceding lemma is true for $K$ any perfect field (and our proof works in
this more general case). The field $\mathbf{F}_{p}(T)$ of rational functions
in one variable over $\mathbf{F}_{p}$ is not perfect, inasmuch as the
variable $T$ is not a $p$th power in $\mathbf{F}_{p}(T)$.

\begin{theorem}
\label{thm:II.4.1}
Let $K$ be a field of characteristic zero or a finite field, let $K'$ be an
extension of finite degree $n$ of $K$, and let $C$ be an algebraically closed
field containing $K$. Then there exist $n$ distinct $K$-isomorphisms of $K'$
into $C$.
\end{theorem}

\begin{proof}
Our assertion is true for any extension field $K'$ of $K$ which is of the
form $K[x]$ with $x\in K'$. In fact, the minimal polynomial $F(X)$ of $x$
over $K$ is then of degree $n$. It has $n$ roots $x_{1},\ldots,x_{n}$ in
$C$, all of which are distinct according to the lemma. For any
$i=1,\ldots,n$ we have then a $K$-isomorphism $\sigma_{i}:K'\to C$ such that
$\sigma_{i}(x)=x_{i}$. We continue by induction on the degree $n$ of $K'$.
Let $x\in K'$, consider the fields $K\subset K[x]\subset K'$ and put
$q=[K[x]:K]$. We may assume $q>1$. We have seen that there are $q$ distinct
$K$-isomorphisms $\sigma_{1},\ldots,\sigma_{q}$ of $K[x]$ into $C$. As
$K[\sigma_{i}(x)]$ and $K[x]$ are isomorphic, it is possible to construct an
extension $K_{i}'$ of $K[\sigma_{i}(x)]$ and an isomorphism
$\tau_{i}:K'\to K_{i}'$ which extends $\sigma_{i}$. Clearly
$K[\sigma_{i}(x)]$ is a field of characteristic zero or a finite field. Since
\[
[K_{i}':K[\sigma_{i}(x)]]=[K':K[x]]=n/q<n,
\]
the induction hypothesis implies that there are $n/q$ distinct
$K[\sigma_{i}(x)]$-isomorphisms $\theta_{ij}$ of $K_{i}'$ into $C$.
Therefore, the $n$ composed mappings $\theta_{ij}\circ\tau_{i}$ provide
$q\cdot n/q=n$ $K$-isomorphisms of $K'$ into $C$. They are distinct, since,
for $i\neq i'$, $\theta_{ij}\circ\tau_{i}$ and $\theta_{i'j'}\circ\tau_{i'}$
differ on $K[x]$ while, for $i=i'$ but $j\neq j'$, $\theta_{ij}$ and
$\theta_{ij'}$ differ on $K_{i}'$.
\end{proof}

Theorem \ref{thm:II.4.1} extends to the case of a perfect field $K$. One
shows that any algebraic extension of a perfect field (in particular
$K[\sigma_{i}(x)]$) is a perfect field. The rest of the proof remains
unchanged.

\begin{statement}{Corollary (``theorem of the primitive element'')}
Let $K$ be a finite field or a field of characteristic zero. Let $K'$ be an
extension of $K$ of finite degree $n$. Then there exists an element $x$ of
$K'$ (called a ``primitive element'') such that $K'=K[x]$.
\end{statement}

\begin{proof}
If $K$ is finite, $K'$ is finite and its multiplicative group $K'^{*}$ is
comprised of the powers of a single element $x$ (\hyperref[sec:1.7]{Chapter I, §7}, Theorem
\ref{thm:I.7.1}, (b)). Thus $K'=K[x]$. Suppose that $K$ is of characteristic
zero and thus an infinite field. According to Theorem \ref{thm:II.4.1} there
are $n$ $K$-isomorphisms $\sigma_{i}$ of $K'$ into an algebraically closed
field $C$ containing $K$. For $i\neq j$ the equation
$\sigma_{i}(y)=\sigma_{j}(y)$ ($y\in K'$) defines a subset $V_{ij}$ of
$K'$, which is clearly a $K$-subspace of the vector space $K'$ and which is
distinct from $K'$ when $\sigma_{i}\neq\sigma_{j}$. Since $K$ is infinite,
linear algebra shows that the union of the $V_{ij}$ is not all of $K'$. Take
$x$ outside this union. The $\sigma_{i}(x)$ are then pairwise distinct, so
that the minimal polynomial $F(X)$ of $x$ over $K$ has at least $n$ distinct
roots (the $\sigma_{i}(x)$) in $C$. Therefore, $d^{\circ}F\geq n$, i.e.\
$[K[x]:K]\geq n$. Since $K[x]\subset K'$ and since $[K':K]=n$, we conclude
that $K'=K[x]$.
\end{proof}

\sechead{5}{Integers in quadratic fields}{INTEGERS IN QUADRATIC FIELDS}

We pause a moment from the development of the general theory to give an
example.

\begin{statement}{Definition}
Any extension field of degree $2$ over the field $\mathbf{Q}$ of rational
numbers is called a \emph{quadratic field}.
\end{statement}

If $K$ is a quadratic field, any element $x\in K-\mathbf{Q}$ is of degree
$2$ over $\mathbf{Q}$, thus is a primitive element of $K$ (i.e.\
$K=\mathbf{Q}[x]$ and $(1,x)$ is a base of $K$ over $\mathbf{Q}$). Let
$F(X)=X^{2}+bX+c$ ($b,c\in\mathbf{Q}$) be the minimal polynomial of such an
element $x\in K$. Solving the quadratic equation $x^{2}+bx+c=0$ gives
$2x=-b\pm\sqrt{b^{2}-4c}$.\footnote{By $\sqrt{b^{2}-4c}$ we mean one of the
two elements of $K$ whose square is $b^{2}-4c$.} Thus
$K=\mathbf{Q}(\sqrt{b^{2}-4c})$. Now
$b^{2}-4c$ is a rational number $u/v=uv/v^{2}$ with $u,v\in\mathbf{Z}$. One
sees that $K=\mathbf{Q}(\sqrt{uv})$ with $u,v\in\mathbf{Z}$. In fact, one
sees that it is possible to write $K=\mathbf{Q}(\sqrt{d})$ where $d$ is a
square-free integer ($d$ is plus or minus a product of distinct primes).
Thus:

\begin{proposition}
\label{prop:II.5.1}
Every quadratic field is of the form $\mathbf{Q}(\sqrt{d})$, where $d$ is a
square-free integer.
\end{proposition}

The element $\sqrt{d}$ is a root of the irreducible polynomial $X^{2}-d$.
This element $\sqrt{d}$ has a conjugate in $K$, and the conjugate has to be
$-\sqrt{d}$. There exists an automorphism $\sigma$ of $K$ which sends
$\sqrt{d}$ to $-\sqrt{d}$ (\hyperref[sec:2.4]{§4}). Any element of $K$ is of the form
$a+b\sqrt{d}$ with $a,b\in\mathbf{Q}$. We have
\begin{equation}
\sigma(a+b\sqrt{d})=a-b\sqrt{d}.
\label{eq:II.5.1}
\end{equation}
We consider the ring $A$ of integers of $K$, i.e.\ the set of $x\in K$ which
are integral over $\mathbf{Z}$ (\hyperref[sec:2.1]{§1}, Corollary \ref{cor:II.1.2} of
Proposition \ref{prop:II.1.1}). If $x\in A$, $\sigma(x)$ is a root of the
same equation of integral dependence as $x$, so $\sigma(x)\in A$. We have
then that $x+\sigma(x)\in A$ and $x\sigma(x)\in A$. But, if
$x=a+b\sqrt{d}$ with $a,b\in\mathbf{Q}$, then, according to \eqref{eq:II.5.1},
\begin{equation}
x+\sigma(x)=2a\in\mathbf{Q}\quad\text{and}\quad
x\cdot\sigma(x)=a^{2}-db^{2}\in\mathbf{Q}.
\label{eq:II.5.2}
\end{equation}
Since $\mathbf{Z}$ is a principal ideal ring and hence integrally closed
(\hyperref[sec:2.2]{§2}, \hyperref[exa:II.2.2]{Example 2}), we see that
\begin{equation}
2a\in\mathbf{Z};\quad a^{2}-db^{2}\in\mathbf{Z}.
\label{eq:II.5.3}
\end{equation}
The conditions \eqref{eq:II.5.3} are necessary in order that
$x=a+b\sqrt{d}$ be integral over $\mathbf{Z}$. They are also sufficient,
since $x$ is a root of $X^{2}-2aX+a^{2}-db^{2}=0$. According to
\eqref{eq:II.5.3}, $(2a)^{2}-d(2b)^{2}\in\mathbf{Z}$. Since
$2a\in\mathbf{Z}$, we have $d(2b)^{2}\in\mathbf{Z}$ too. On the other hand,
$d$ is square-free, so, if $2b$ were not an integer, its denominator would
have to include a prime factor $p$. This prime factor would have to appear as
$p^{2}$ in the denominator of $(2b)^{2}$. Multiplication by $d$ would not
send $(2b)^{2}$ into $\mathbf{Z}$. We may conclude that $2b\in\mathbf{Z}$.

In brief, we may take $a=u/2$, $b=v/2$ with $u,v\in\mathbf{Z}$. Condition
\eqref{eq:II.5.3} becomes:
\begin{equation}
u^{2}-dv^{2}\in4\mathbf{Z}.
\label{eq:II.5.4}
\end{equation}
If $v$ is even, \eqref{eq:II.5.4} shows that $u$ is even too. In this case,
$a,b\in\mathbf{Z}$. If $v$ is odd, then $v^{2}\equiv1\bmod4$. The
possibilities mod $4$ for $u^{2}$ are $0$ and $1$ (the only squares mod $4$).
Since $d$ is square-free, it is not a multiple of $4$. Necessarily
$u^{2}\equiv1\bmod4$ and $d\equiv1\bmod4$. We have proved the following:

\begin{theorem}
\label{thm:II.5.1}
Let $K=\mathbf{Q}(\sqrt{d})$ be a quadratic field with $d\in\mathbf{Z}$
square-free (therefore $\not\equiv0\bmod4$).
\begin{alist}
\item If $d\equiv2$ or $d\equiv3\bmod4$, the ring $A$ of integers of $K$
consists of all elements of the form $a+b\sqrt{d}$ with
$a,b\in\mathbf{Z}$.
\item If $d\equiv1\bmod4$, $A$ consists of all elements of the form
$\frac{1}{2}(u+v\sqrt{d})$ with $u$ and $v\in\mathbf{Z}$ of the same parity.
\end{alist}
\end{theorem}

In the case that $d\equiv2$ or $3\bmod4$, $(1,\sqrt{d})$ is a base for $A$
as a $\mathbf{Z}$-module. If $d\equiv1\bmod4$, $(1,\frac{1}{2}(1+\sqrt{d}))$
is a $\mathbf{Z}$-module base for $A$. Indeed, by (b), $1$ and
$\frac{1}{2}(1+\sqrt{d})$ belong to $A$. Conversely, to show that
$\frac{1}{2}(u+v\sqrt{d})$ (with $u,v\in\mathbf{Z}$ of the same parity) is
expressible as a $\mathbf{Z}$-linear combination of $1$ and
$\frac{1}{2}(1+\sqrt{d})$, one may, by subtracting
$\frac{1}{2}(1+\sqrt{d})$, reduce the problem to the case where $u$ and $v$
are even. In this case
\[
\frac{1}{2}(u+v\sqrt{d})=\left(\frac{u}{2}-\frac{v}{2}\right)\cdot1
+v\cdot\frac{1}{2}(1+\sqrt{d}).
\]

We conclude with some terminology. If $d>0$, $\mathbf{Q}(\sqrt{d})$ is
called a \emph{real quadratic field} (there exists a subfield of
$\mathbf{R}$ conjugate to $\mathbf{Q}(\sqrt{d})$ over $\mathbf{Q}$). If
$d<0$, then $\mathbf{Q}(\sqrt{d})$ is called an \emph{imaginary quadratic
field}.
