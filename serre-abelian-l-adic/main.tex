%% common pieces ---------------------------------------------------------
\newbibmacro{serre-names}{%
  \ifnameundef{author}
    {\ifnameundef{editor}{}{\printnames{editor}}}
    {\printnames{author}}}
\newcommand{\serreafternames}{\addspace\textemdash\addspace}
%% journal / volume / year / pages  (two punctuations: main vs supp)
\newbibmacro{serre-jvp}{%
  \usebibmacro{journal}%
  \ifcategory{supp}{\setunit*{\addspace}}{\setunit*{\addcomma\addspace}}%
  \printfield{volume}%
  \setunit*{\addcomma\addspace}%
  \printfield{year}%
  \setunit*{\addcomma\addspace}%
  \printfield{pages}}
\newbibmacro{serre-booktail}{%
  \printfield{series}%
  \setunit*{\addcomma\addspace}%
  \printlist{publisher}%
  \setunit*{\addcomma\addspace}%
  \printlist{location}%
  \setunit*{\addcomma\addspace}%
  \printfield{year}}

%% ---------- ToC typesetting (no dot leaders, as the original) ----------
%% Each entry carries its own label (4th/3rd arg); the page number links to
%% that label rather than to hyperref's per-page anchor, because hyperref
%% omits the "page.<folio>" destination on some pages (measured: II-9, III-22,
%% III-26, IV-14, IV-28), which left those TOC page numbers as dead links.
\makeatletter
\newcommand{\fml}[3]{%
  \addvspace{1.5\baselineskip}\noindent
  {\bfseries #1}\nobreak\hfill\hyperpage@serre{#2}{#3}\par}
\newcommand{\chlc}[4]{%  chapter
  \addvspace{1.6\baselineskip}\noindent
  {\bfseries #1\quad #2}\nobreak\hfill\hyperpage@serre{#3}{#4}\par}
\newcommand{\chls}[4]{%  §-section
  \nobreak\vspace{1.0\baselineskip}\noindent
  {\bfseries §#1\quad #2}\nobreak\hfill\hyperpage@serre{#3}{#4}\par}
\newcommand{\chlapp}[4]{% appendix (chapter level)
  \nobreak\vspace{1.0\baselineskip}\noindent
  {\bfseries #1: #2}\nobreak\hfill\hyperpage@serre{#3}{#4}\par}
\newcommand{\chlu}[4]{%  subsection
  \par\noindent\hspace{2em}%
  {\itshape #1\quad #2}\nobreak\hfill\hyperpage@serre{#3}{#4}\par}
%% page number -> the entry's own label (falls back to the page anchor if the
%% label is somehow undefined, so a TOC line never renders "??").
\newcommand{\hyperpage@serre}[2]{%
  \ifx\\#1\\\else
    \@ifundefined{r@#2}{\hyperlink{page.#1}{#1}}{\hyperref[#2]{#1}}%
  \fi}
\makeatother
